function CompleteMission  [FUNC_Public | FUNC_BlueprintCallable | FUNC_BlueprintEvent]
            self.IsCompleted = true
            NetPushModelHelpers.MarkPropertyDirtyFromRepIndex(self, 2, "IsCompleted")
            OnRep_IsCompleted()
            K2Node_SwitchInteger_CmpSuccess = KismetMathLibrary.NotEqual_IntInt(self.TaskListIndex, 0)
            if not (K2Node_SwitchInteger_CmpSuccess) goto @155
            K2Node_SwitchInteger_CmpSuccess = KismetMathLibrary.NotEqual_IntInt(self.TaskListIndex, 1)
            if not (K2Node_SwitchInteger_CmpSuccess) goto @262
            goto @364
    @155:   self.OwningManager.Actor.FlushNetDormancy()
            self.OwningManager.self.Task1Completed = true
            NetPushModelHelpers.MarkPropertyDirtyFromRepIndex(self.OwningManager, 13, "Task1Completed")
            goto @364
    @262:   self.OwningManager.Actor.FlushNetDormancy()
            self.OwningManager.self.Task2Completed = true
            NetPushModelHelpers.MarkPropertyDirtyFromRepIndex(self.OwningManager, 14, "Task2Completed")
    @364:   return
